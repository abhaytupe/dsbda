import java.util.*;

public class WordCount {

    public static void main(String[] args) {

        String text = "hadoop mapreduce big data hadoop big data";

        String[] words = text.split(" ");

        HashMap<String, Integer> map = new HashMap<>();

        for(String word : words) {

            if(map.containsKey(word)) {
                map.put(word, map.get(word) + 1);
            } 
            else {
                map.put(word, 1);
            }
        }

        System.out.println("Word Count Output:");

        for(String word : map.keySet()) {
            System.out.println(word + " : " + map.get(word));
        }
    }
}

public class WeatherAnalysis {

    public static void main(String[] args) {

        double temperature[] = {30, 32, 35, 31, 29};

        double dewPoint[] = {20, 21, 19, 18, 20};

        double windSpeed[] = {10, 12, 8, 9, 11};

        double tempAvg = 0;
        double dewAvg = 0;
        double windAvg = 0;

        for(int i = 0; i < temperature.length; i++) {

            tempAvg += temperature[i];

            dewAvg += dewPoint[i];

            windAvg += windSpeed[i];
        }

        tempAvg = tempAvg / temperature.length;

        dewAvg = dewAvg / dewPoint.length;

        windAvg = windAvg / windSpeed.length;

        System.out.println("Average Temperature: " + tempAvg);

        System.out.println("Average Dew Point: " + dewAvg);

        System.out.println("Average Wind Speed: " + windAvg);
    }
}
STEP 3 — Compile
javac WeatherAnalysis.java
STEP 4 — Run
java WeatherAnalysis
OUTPUT
Average Temperature: 31.4
Average Dew Point: 19.6
Average Wind Speed: 10.0
